// Extract all placeholders from a template text
function extractPlaceholders(text) {
  const matches = text.match(/\[([^\]]+)\]/g) || [];
  return [...new Set(matches)];
}

// Fill placeholders with values
function fillPlaceholders(text, values) {
  let result = text;
  Object.entries(values).forEach(([key, val]) => {
    result = result.split(key).join(val || key);
  });
  return result;
}

// ─── TOAST HOOK ───────────────────────────────────────────
function useToast() {
  const [toast, setToast] = useState(null);
  const show = useCallback((msg, color = C.accent) => {
    setToast({ msg, color });
    setTimeout(() => setToast(null), 2400);
  }, []);
  return { toast, show };
}

function Toast({ toast }) {
  if (!toast) return null;
  return (
    <View style={[ss.toast, { backgroundColor: toast.color }]}>
      <Text style={ss.toastTxt}>{toast.msg}</Text>
    </View>
  );
}

// ─── PRICE TICKER ─────────────────────────────────────────
function PriceTicker() {
  const [prices, setPrices] = useState(null);
  const [loading, setLoading] = useState(true);
  const [lastUpdated, setLastUpdated] = useState(null);

  const fetchPrices = useCallback(async () => {
    try {
      const res = await fetch(
        'https://api.coingecko.com/api/v3/simple/price?ids=bitcoin,ethereum,tether,binancecoin,solana&vs_currencies=usd,inr&include_24hr_change=true',
        { headers: { 'Accept': 'application/json' } }
      );
      const data = await res.json();
      setPrices(data);
      setLastUpdated(new Date());
    } catch {
      // silently fail -- prices just stay as last known
    }
    setLoading(false);
  }, []);

  useEffect(() => {
    fetchPrices();
    const interval = setInterval(fetchPrices, 60000);
    return () => clearInterval(interval);
  }, []);

  const COINS = [
    { id: 'bitcoin',     symbol: 'BTC', emoji: '₿' },
    { id: 'ethereum',    symbol: 'ETH', emoji: 'Ξ' },
    { id: 'tether',      symbol: 'USDT', emoji: '₮' },
    { id: 'binancecoin', symbol: 'BNB', emoji: '🔶' },
    { id: 'solana',      symbol: 'SOL', emoji: '◎' },
  ];

  if (loading) return (
    <View style={ss.tickerLoading}>
      <ActivityIndicator size="small" color={C.accent} />
      <Text style={ss.tickerLoadTxt}>Fetching live prices...</Text>
    </View>
  );

  return (
    <View style={ss.tickerWrap}>
      <View style={ss.tickerHeader}>
        <Text style={ss.tickerTitle}>💹 Live Prices</Text>
        <TouchableOpacity onPress={fetchPrices} style={ss.refreshBtn}>
          <Text style={ss.refreshBtnTxt}>↻ Refresh</Text>
        </TouchableOpacity>
      </View>
      {lastUpdated && (
        <Text style={ss.tickerUpdated}>
          Updated {lastUpdated.toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit' })}
        </Text>
      )}
      <ScrollView horizontal showsHorizontalScrollIndicator={false} style={ss.tickerScroll}>
        {COINS.map(coin => {
          const d = prices?.[coin.id];
          if (!d) return null;
          const change = d.usd_24h_change;
          const isUp = change >= 0;
          const changeColor = isUp ? C.green : C.red;
          return (
            <View key={coin.id} style={ss.priceCard}>
              <Text style={ss.coinEmoji}>{coin.emoji}</Text>
              <Text style={ss.coinSymbol}>{coin.symbol}</Text>
              <Text style={ss.coinUsd}>${d.usd?.toLocaleString('en-US', { maximumFractionDigits: 2 })}</Text>
              <Text style={ss.coinInr}>₹{d.inr?.toLocaleString('en-IN', { maximumFractionDigits: 0 })}</Text>
              {change !== undefined && (
                <View style={[ss.changeBadge, { backgroundColor: changeColor + '22' }]}>
                  <Text style={[ss.changeTxt, { color: changeColor }]}>
                    {isUp ? '▲' : '▼'} {Math.abs(change).toFixed(2)}%
                  </Text>
                </View>
              )}
            </View>
          );
        })}
      </ScrollView>
    </View>
  );
}

// ─── PLACEHOLDER MODAL ────────────────────────────────────
function PlaceholderModal({ visible, template, onClose, onCopy }) {
  const placeholders = template ? extractPlaceholders(template.text) : [];
  const [values, setValues] = useState({});

  useEffect(() => {
    if (visible) {
      const init = {};
      (template ? extractPlaceholders(template.text) : []).forEach(p => { init[p] = ''; });
      setValues(init);
    }
  }, [visible, template]);

  if (!template) return null;

  const hasPlaceholders = placeholders.length > 0;
  const preview = fillPlaceholders(template.text, values);

  return (
    <Modal visible={visible} transparent animationType="slide">
      <KeyboardAvoidingView style={ss.modalOv} behavior={Platform.OS === 'ios' ? 'padding' : 'height'}>
        <View style={[ss.modalSheet, { maxHeight: '90%' }]}>
          <View style={ss.sheetHandle} />
          <Text style={ss.sheetTitle}>📋 {hasPlaceholders ? 'Fill Placeholders' : 'Copy Template'}</Text>
          <Text style={ss.sheetSub}>{template.title}</Text>

          <ScrollView showsVerticalScrollIndicator={false} style={{ maxHeight: 400 }}>
            {hasPlaceholders ? (
              <>
                <Text style={ss.inputLbl}>FILL IN THE DETAILS</Text>
                {placeholders.map(ph => (
                  <View key={ph} style={{ marginBottom: 10 }}>
                    <Text style={ss.phLabel}>{ph}</Text>
                    <TextInput
                      style={ss.sheetInput}
                      placeholder={`Enter value for ${ph}`}
                      placeholderTextColor={C.t3}
                      value={values[ph] || ''}
                      onChangeText={v => setValues(prev => ({ ...prev, [ph]: v }))}
                    />
                  </View>
                ))}
                <Text style={ss.inputLbl}>PREVIEW</Text>
                <View style={ss.previewBox}>
                  <Text style={ss.previewTxt}>{preview}</Text>
                </View>
              </>
            ) : (
              <View style={ss.previewBox}>
                <Text style={ss.previewTxt}>{template.text}</Text>
              </View>
            )}
          </ScrollView>

          <View style={ss.sheetBtns}>
            <TouchableOpacity style={ss.savBtn} onPress={() => onCopy(preview)}>
              <Text style={ss.savBtnTxt}>📋 Copy & Use</Text>
            </TouchableOpacity>
            <TouchableOpacity style={ss.skipBtn} onPress={onClose}>
              <Text style={ss.skipBtnTxt}>Cancel</Text>
            </TouchableOpacity>
          </View>
        </View>
      </KeyboardAvoidingView>
    </Modal>
  );
}

// ─── CUSTOM TEMPLATE MODAL ────────────────────────────────
function CustomTemplateModal({ visible, editing, onClose, onSave }) {
  const [title, setTitle] = useState('');
  const [text, setText] = useState('');
  const [tags, setTags] = useState('');

  useEffect(() => {
    if (editing) {
      setTitle(editing.title || '');
      setText(editing.text || '');
      setTags((editing.tags || []).join(', '));
    } else {
      setTitle(''); setText(''); setTags('');
    }
  }, [editing, visible]);

  const handleSave = () => {
    if (!title.trim() || !text.trim()) {
      Alert.alert('Missing Info', 'Please add a title and response text.');
      return;
    }
    const tagList = tags.split(',').map(t => t.trim().toLowerCase()).filter(Boolean);
    onSave({
      id: editing?.id || 'custom_' + Date.now(),
      title: title.trim(),
      text: text.trim(),
      tags: tagList,
      category: '📝 My Templates',
      isCustom: true,
    });
    onClose();
  };

  return (
    <Modal visible={visible} transparent animationType="slide">
      <KeyboardAvoidingView style={ss.modalOv} behavior={Platform.OS === 'ios' ? 'padding' : 'height'}>
        <View style={[ss.modalSheet, { maxHeight: '92%' }]}>
          <View style={ss.sheetHandle} />
          <Text style={ss.sheetTitle}>{editing ? '✏️ Edit Template' : '📝 New Custom Template'}</Text>
          <ScrollView showsVerticalScrollIndicator={false}>
            <Text style={ss.inputLbl}>TITLE</Text>
            <TextInput style={ss.sheetInput} placeholder="e.g. Weekend Delay Notice"
              placeholderTextColor={C.t3} value={title} onChangeText={setTitle} />

            <Text style={ss.inputLbl}>RESPONSE TEXT</Text>
            <Text style={ss.phHint}>Tip: Use [Customer Name], [TICKET_ID], [AMOUNT] for auto-fill placeholders</Text>
            <TextInput
              style={[ss.sheetInput, { minHeight: 140, textAlignVertical: 'top', paddingTop: 12 }]}
              placeholder="Type your response here..."
              placeholderTextColor={C.t3}
              value={text}
              onChangeText={setText}
              multiline
            />

            <Text style={ss.inputLbl}>TAGS (comma separated)</Text>
            <TextInput style={ss.sheetInput} placeholder="e.g. delay, weekend, notice"
              placeholderTextColor={C.t3} value={tags} onChangeText={setTags} />
          </ScrollView>

          <View style={[ss.sheetBtns, { marginTop: 16 }]}>
            <TouchableOpacity style={ss.savBtn} onPress={handleSave}>
              <Text style={ss.savBtnTxt}>{editing ? 'Save Changes' : '+ Save Template'}</Text>
            </TouchableOpacity>
            <TouchableOpacity style={ss.skipBtn} onPress={onClose}>
              <Text style={ss.skipBtnTxt}>Cancel</Text>
            </TouchableOpacity>
          </View>
        </View>
      </KeyboardAvoidingView>
    </Modal>
  );
}

      if (next.has(id)) { next.delete(id); showToast('Removed from Favourites'); }
      else { next.add(id); showToast('⭐ Added to Favourites!'); }
      store.set(K.favourites, [...next]);
      return next;
    });
  }, [showToast]);

  const saveCustomTemplate = useCallback(async (tpl) => {
    setCustomTemplates(prev => {
      const existing = prev.findIndex(t => t.id === tpl.id);
      const updated = existing >= 0
        ? prev.map(t => t.id === tpl.id ? tpl : t)
        : [...prev, tpl];
      store.set(K.customTemplates, updated);
      return updated;
    });
    showToast(editingTemplate ? '✓ Template updated!' : '✓ Custom template saved!');
  }, [editingTemplate, showToast]);

  const deleteCustomTemplate = useCallback((id) => {
    Alert.alert('Delete Template', 'Delete this custom template?', [
      { text: 'Cancel', style: 'cancel' },
      {
        text: 'Delete', style: 'destructive', onPress: () => {
          setCustomTemplates(prev => {
            const updated = prev.filter(t => t.id !== id);
            store.set(K.customTemplates, updated);
            return updated;
          });
          showToast('Template deleted', C.red);
        }
      },
    ]);
  }, [showToast]);

  // Build full template list including custom
  const allTemplates = [
    ...Object.entries(BUILT_IN_TEMPLATES).flatMap(([c, items]) =>
      items.map(t => ({ ...t, category: c }))),
    ...customTemplates,
  ];

  const favTemplates = allTemplates.filter(t => favourites.has(t.id));

  const CATEGORIES = [
    'All',
    ...(favTemplates.length > 0 ? ['⭐ Favourites'] : []),
    ...(customTemplates.length > 0 ? ['📝 My Templates'] : []),
    ...Object.keys(BUILT_IN_TEMPLATES),
  ];

  const filtered = allTemplates.filter(t => {
    if (cat === '⭐ Favourites') return favourites.has(t.id);
    if (cat === '📝 My Templates') return t.isCustom;
    const matchCat = cat === 'All' || t.category === cat;
    const q = search.toLowerCase();
    return matchCat && (!q || t.title.toLowerCase().includes(q) ||
      t.text.toLowerCase().includes(q) || t.tags?.some(g => g.includes(q)));
  }).filter(t => {
    if (cat === 'All' || cat === '⭐ Favourites' || cat === '📝 My Templates') return true;
    const q = search.toLowerCase();
    return !q || t.title.toLowerCase().includes(q) ||
      t.text.toLowerCase().includes(q) || t.tags?.some(g => g.includes(q));
  });

  // When copy tapped -- show placeholder modal first
  const handleCopyTap = useCallback((t) => {
    setPhTemplate(t);
    setPhModal(true);
  }, []);

  const handleConfirmedCopy = useCallback((finalText) => {
    Clipboard.setString(finalText);
    if (phTemplate) {
      setCopiedId(phTemplate.id);
      setTimeout(() => setCopiedId(null), 2000);
      setPending({ ...phTemplate, text: finalText });
      setTicketId(''); setLogStatus('Used'); setLogNote('');
    }
    setPhModal(false);
    setLogModal(true);
    showToast('✓ Copied to clipboard!');
  }, [phTemplate, showToast]);

  const doLog = async () => {
    if (!pending) return;
    const entry = {
      id: Date.now().toString(),
      templateId: pending.id,
      templateTitle: pending.title,
      category: pending.category,
      ticketId, status: logStatus, note: logNote,
      timestamp: new Date().toISOString(),
      text: pending.text,
    };
    await onLog(entry);
    setLogModal(false);
    showToast('✓ Usage logged!');
  };

  const renderItem = ({ item: t }) => {
    const isExp = expanded === t.id;
    const isCop = copiedId === t.id;
    const isFav = favourites.has(t.id);

    return (
      <TouchableOpacity style={[ss.card, t.isCustom && ss.customCard]}
        onPress={() => setExpanded(isExp ? null : t.id)} activeOpacity={0.85}>
        <View style={ss.cardHead}>
          <View style={ss.cardLeft}>
            <TouchableOpacity onPress={() => toggleFav(t.id)} style={ss.favBtn}>
              <Text style={[ss.favStar, isFav && ss.favStarOn]}>⭐</Text>
            </TouchableOpacity>
            <View style={{ flex: 1 }}>
              <Text style={ss.cardTitle}>
                {t.isCustom ? '📝 ' : ''}{t.title}
              </Text>
              <Text style={ss.cardCat}>{CAT_ICONS[t.category] || '💬'} {t.category}</Text>
            </View>
          </View>
          <View style={ss.cardBtns}>
            {t.isCustom && (
              <TouchableOpacity style={ss.editBtn} onPress={() => {
                setEditingTemplate(t); setShowCustomModal(true);
              }}>
                <Text style={ss.editBtnTxt}>✏️</Text>
              </TouchableOpacity>
            )}
            <TouchableOpacity style={[ss.copyBtn, isCop && ss.copyBtnOn]} onPress={() => handleCopyTap(t)}>
              <Text style={[ss.copyBtnTxt, isCop && ss.copyBtnTxtOn]}>{isCop ? '✓' : 'Copy'}</Text>
            </TouchableOpacity>
          </View>
        </View>

        <Text style={ss.cardPrev} numberOfLines={isExp ? undefined : 2}>{t.text}</Text>

        {t.tags?.length > 0 && (
          <View style={ss.tagRow}>
            {t.tags.map(g => <View key={g} style={ss.tag}><Text style={ss.tagTxt}>#{g}</Text></View>)}
            <Text style={ss.expandHint}>{isExp ? '▲ less' : '▼ more'}</Text>
          </View>
        )}

        {t.isCustom && (
          <TouchableOpacity onPress={() => deleteCustomTemplate(t.id)} style={ss.deleteCustomBtn}>
            <Text style={ss.deleteCustomTxt}>🗑️ Delete</Text>
          </TouchableOpacity>
        )}
      </TouchableOpacity>
    );
  };

  return (
    <View style={{ flex: 1 }}>
      {/* Live Prices */}
      <PriceTicker />

      {/* Search */}
      <View style={ss.searchBar}>
        <Text style={ss.searchIco}>🔍</Text>
        <TextInput style={ss.searchIn} placeholder="Search templates, tags..."
          placeholderTextColor={C.t3} value={search} onChangeText={setSearch} />
        {!!search && <TouchableOpacity onPress={() => setSearch('')}><Text style={ss.clearBtn}>✕</Text></TouchableOpacity>}
      </View>

      {/* Categories */}
      <ScrollView horizontal showsHorizontalScrollIndicator={false}
        style={ss.catScroll} contentContainerStyle={ss.catScrollIn}>
        {CATEGORIES.map(c => (
          <TouchableOpacity key={c} style={[ss.catBtn, cat === c && ss.catBtnOn]} onPress={() => setCat(c)}>
            <Text style={[ss.catBtnTxt, cat === c && ss.catBtnTxtOn]}>
              {c === 'All' ? '📋 All' : c}
            </Text>
          </TouchableOpacity>
        ))}
      </ScrollView>

      {/* Count + New Template button */}
      <View style={ss.listTopRow}>
        <Text style={ss.countTxt}>{filtered.length} template{filtered.length !== 1 ? 's' : ''}</Text>
        <TouchableOpacity style={ss.newTplBtn} onPress={() => { setEditingTemplate(null); setShowCustomModal(true); }}>
          <Text style={ss.newTplBtnTxt}>+ New</Text>
        </TouchableOpacity>
      </View>

      <FlatList data={filtered} keyExtractor={i => i.id} renderItem={renderItem}
        contentContainerStyle={ss.listIn} showsVerticalScrollIndicator={false}
        keyboardShouldPersistTaps="handled"
        ListEmptyComponent={
          <View style={ss.emptyMini}>
            <Text style={ss.emptyMiniTxt}>No templates found</Text>
          </View>
        }
      />

      {/* Placeholder Fill Modal */}
      <PlaceholderModal
        visible={phModal}
        template={phTemplate}
        onClose={() => setPhModal(false)}
        onCopy={handleConfirmedCopy}
      />

      {/* Custom Template Modal */}
      <CustomTemplateModal
        visible={showCustomModal}
        editing={editingTemplate}
        onClose={() => { setShowCustomModal(false); setEditingTemplate(null); }}
        onSave={saveCustomTemplate}
      />

      {/* Log Modal */}
      <Modal visible={logModal} transparent animationType="slide">
        <KeyboardAvoidingView style={ss.modalOv} behavior={Platform.OS === 'ios' ? 'padding' : 'height'}>
          <View style={ss.modalSheet}>
            <View style={ss.sheetHandle} />
            <Text style={ss.sheetTitle}>📋 Log Usage</Text>
            {pending && <Text style={ss.sheetSub}>{pending.title}</Text>}
            <Text style={ss.inputLbl}>ZENDESK TICKET ID</Text>
            <TextInput style={ss.sheetInput} placeholder="e.g. ZD-12345"
              placeholderTextColor={C.t3} value={ticketId} onChangeText={setTicketId} />
            <Text style={ss.inputLbl}>STATUS</Text>
            <ScrollView horizontal showsHorizontalScrollIndicator={false}>
              {STATUS_OPTIONS.map(s => (
                <TouchableOpacity key={s} style={[ss.statusChip, logStatus === s && ss.statusChipOn]} onPress={() => setLogStatus(s)}>
                  <Text style={[ss.statusChipTxt, logStatus === s && ss.statusChipTxtOn]}>{s}</Text>
                </TouchableOpacity>
              ))}
            </ScrollView>
            <Text style={ss.inputLbl}>NOTE (OPTIONAL)</Text>
            <TextInput style={[ss.sheetInput, { minHeight: 50 }]} placeholder="Any notes about this response..."
              placeholderTextColor={C.t3} value={logNote} onChangeText={setLogNote} multiline />
            <View style={ss.sheetBtns}>
              <TouchableOpacity style={ss.savBtn} onPress={doLog}><Text style={ss.savBtnTxt}>Save Log</Text></TouchableOpacity>
              <TouchableOpacity style={ss.skipBtn} onPress={() => setLogModal(false)}><Text style={ss.skipBtnTxt}>Skip</Text></TouchableOpacity>
            </View>
          </View>
        </KeyboardAvoidingView>
      </Modal>
    </View>
  );
}

// ─── AI SCREEN ────────────────────────────────────────────
function AIScreen({ showToast }) {
  const [query, setQuery] = useState('');
  const [result, setResult] = useState('');
  const [loading, setLoading] = useState(false);
  const [copied, setCopied] = useState(false);
  const [settings, setSettings] = useState({ claudeKey: '' });

  useEffect(() => { store.get(K.settings, {}).then(setSettings); }, []);

  const SYSTEM = `You are a customer support specialist for an Indian crypto exchange. Help support agents write professional, compliant, empathetic responses.

Rules:
- Comply with PMLA, IT Act Section 115BBH/194S, FIU-IND VASP guidelines
- Empathetic and professional tone
- Clear and jargon-free
- Ready to copy-paste directly

Provide TWO variations:

**Brief Response:** (under 80 words)
[short version]

**Detailed Response:** (full version)
[detailed version]`;

  const ask = async () => {
    if (!query.trim()) return;
    if (!settings.claudeKey) { showToast('⚠️ Add Claude API key in Settings', C.warn); return; }
    setLoading(true); setResult('');
    try {
      const r = await fetch('https://api.anthropic.com/v1/messages', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'x-api-key': settings.claudeKey, 'anthropic-version': '2023-06-01' },
        body: JSON.stringify({ model: 'claude-sonnet-4-20250514', max_tokens: 1000, system: SYSTEM, messages: [{ role: 'user', content: query }] }),
      });
      const d = await r.json();
      if (d.error) throw new Error(d.error.message);
      setResult(d.content?.map(c => c.text || '').join('') || 'No response.');
    } catch (e) {
      setResult('Error: ' + e.message + '\n\nPlease add your Claude API key in Settings tab.');
    }
    setLoading(false);
  };

  return (
    <KeyboardAvoidingView style={{ flex: 1 }} behavior={Platform.OS === 'ios' ? 'padding' : 'height'}>
      <ScrollView style={{ flex: 1 }} contentContainerStyle={ss.aiContent} keyboardShouldPersistTaps="handled">
        <View style={ss.infoBanner}>
          <Text style={ss.infoTxt}>◈ Uses Claude AI -- add your free API key in Settings. Get it at console.anthropic.com</Text>
        </View>
        <Text style={ss.secLabel}>QUICK PROMPTS</Text>
        <ScrollView horizontal showsHorizontalScrollIndicator={false} style={{ marginBottom: 14 }}>
          {QUICK_PROMPTS.map(p => (
            <TouchableOpacity key={p} style={ss.qChip} onPress={() => setQuery(p)}>
              <Text style={ss.qChipTxt}>{p}</Text>
            </TouchableOpacity>
          ))}
        </ScrollView>
        <Text style={ss.secLabel}>YOUR QUERY</Text>
        <TextInput style={ss.aiInput} placeholder="Describe the customer issue..."
          placeholderTextColor={C.t3} value={query} onChangeText={setQuery} multiline numberOfLines={4} textAlignVertical="top" />
        <TouchableOpacity style={[ss.aiSubmit, (!query.trim() || loading) && { opacity: 0.5 }]} onPress={ask} disabled={!query.trim() || loading}>
          {loading ? <ActivityIndicator color={C.white} /> : <Text style={ss.aiSubmitTxt}>✨ Generate Response with Claude</Text>}
        </TouchableOpacity>
        {!!result && (
          <View style={ss.resultBox}>
            <View style={ss.resultHead}>
              <Text style={ss.resultTitle}>AI Generated Response</Text>
              <TouchableOpacity style={[ss.copyBtn, copied && ss.copyBtnOn]} onPress={() => {
                Clipboard.setString(result); setCopied(true);
                setTimeout(() => setCopied(false), 2000);
                showToast('✓ Copied!');
              }}>
                <Text style={[ss.copyBtnTxt, copied && ss.copyBtnTxtOn]}>{copied ? '✓' : 'Copy'}</Text>
              </TouchableOpacity>
            </View>
            <Text style={ss.resultTxt}>{result}</Text>
          </View>
        )}
      </ScrollView>
    </KeyboardAvoidingView>
  );
}

// ─── TOOLS SCREEN (Blockchain TX + Zendesk) ───────────────
function ToolsScreen({ showToast }) {
  const [toolTab, setToolTab] = useState('tx');

  // TX Checker state
  const [txid, setTxid] = useState('');
  const [txNetwork, setTxNetwork] = useState('BTC');
  const [txResult, setTxResult] = useState(null);
  const [txLoading, setTxLoading] = useState(false);
  const [txError, setTxError] = useState('');

  // Zendesk state
  const [zdTicket, setZdTicket] = useState('');
  const [zdResult, setZdResult] = useState(null);
  const [zdLoading, setZdLoading] = useState(false);
  const [zdError, setZdError] = useState('');
  const [zdSettings, setZdSettings] = useState({ zdSubdomain: '', zdEmail: '', zdToken: '' });

  useEffect(() => { store.get(K.settings, {}).then(setZdSettings); }, []);

  const TX_NETWORKS = ['BTC', 'ETH / ERC20', 'TRC20 (TRON)'];

  const checkTx = async () => {
    if (!txid.trim()) return;
    setTxLoading(true); setTxResult(null); setTxError('');
    try {
      let data = null;
      if (txNetwork === 'BTC') {
        const res = await fetch(`https://blockstream.info/api/tx/${txid.trim()}`);
        if (!res.ok) throw new Error('Transaction not found. Check the TxID and network.');
        const d = await res.json();
        const confirmed = d.status?.confirmed;
        const blockHeight = d.status?.block_height;
        // Get current block height
        const tipRes = await fetch('https://blockstream.info/api/blocks/tip/height');
        const tipHeight = tipRes.ok ? await tipRes.json() : null;
        const confirmations = confirmed && tipHeight && blockHeight ? tipHeight - blockHeight + 1 : 0;
        const amountBTC = d.vout?.reduce((s, o) => s + (o.value || 0), 0) / 1e8;
        data = {
          network: 'Bitcoin (BTC)',
          status: confirmed ? '✅ Confirmed' : '⏳ Pending / Unconfirmed',
          confirmations: confirmed ? confirmations : 0,
          required: 3,
          amount: amountBTC?.toFixed(8) + ' BTC',
          fee: d.fee ? (d.fee / 1e8).toFixed(8) + ' BTC' : 'N/A',
          txid: txid.trim(),
        };
      } else if (txNetwork === 'ETH / ERC20') {
        const res = await fetch(`https://api.blockchair.com/ethereum/transactions?q=hash(${txid.trim()})`);
        if (!res.ok) throw new Error('Could not fetch from Blockchair. Try again.');
        const d = await res.json();
        const tx = d.data?.[0];
        if (!tx) throw new Error('Transaction not found on Ethereum network.');
        data = {
          network: 'Ethereum (ETH/ERC20)',
          status: tx.block_id ? '✅ Confirmed' : '⏳ Pending',
          confirmations: tx.block_id ? (d.context?.state - tx.block_id + 1) : 0,
          required: 12,
          amount: tx.value ? (parseFloat(tx.value) / 1e18).toFixed(6) + ' ETH' : 'Token Transfer',
          fee: tx.fee ? (parseFloat(tx.fee) / 1e18).toFixed(6) + ' ETH' : 'N/A',
          txid: txid.trim(),
        };
      } else {
        // TRON / TRC20
        const res = await fetch(`https://apilist.tronscanapi.com/api/transaction-info?hash=${txid.trim()}`);
        if (!res.ok) throw new Error('Could not connect to TRON network. Try again.');
        const d = await res.json();
        if (!d.hash) throw new Error('Transaction not found on TRON network.');
        data = {
          network: 'TRON (TRC20/USDT)',
          status: d.confirmed ? '✅ Confirmed' : '⏳ Pending',
          confirmations: d.confirmed ? 'Confirmed' : 0,
          required: 20,
          amount: d.tokenTransferInfo?.amount_str
            ? (parseInt(d.tokenTransferInfo.amount_str) / 1e6).toFixed(2) + ' ' + (d.tokenTransferInfo.symbol || 'TRX')
            : (d.amount / 1e6)?.toFixed(2) + ' TRX',
          fee: 'Energy/Bandwidth',
          txid: txid.trim(),
        };
      }
      setTxResult(data);
    } catch (e) {
      setTxError(e.message || 'Failed to fetch transaction. Check TxID and network.');
    }
    setTxLoading(false);
  };

  const copyTxReply = () => {
    if (!txResult) return;
    const txt = `We have checked your transaction on the blockchain:\n\n` +
      `• Network: ${txResult.network}\n` +
      `• Status: ${txResult.status}\n` +
      `• Confirmations: ${txResult.confirmations} / ${txResult.required} required\n` +
      `• Amount: ${txResult.amount}\n` +
      `• TxID: ${txResult.txid.slice(0, 20)}...\n\n` +
      (txResult.confirmations < txResult.required
        ? `Your transaction is still awaiting confirmations. Please allow some more time.`
        : `Your transaction is fully confirmed on the blockchain. If funds have not been credited, please contact us with this transaction ID.`);
    Clipboard.setString(txt);
    showToast('✓ TX reply copied!');
  };

  const checkZendesk = async () => {
    if (!zdTicket.trim()) return;
    if (!zdSettings.zdSubdomain || !zdSettings.zdEmail || !zdSettings.zdToken) {
      setZdError('Please add your Zendesk subdomain, email and API token in Settings first.');
      return;
    }
    setZdLoading(true); setZdResult(null); setZdError('');
    try {
      const ticketNum = zdTicket.trim().replace(/[^0-9]/g, '');
      const auth = btoa(`${zdSettings.zdEmail}/token:${zdSettings.zdToken}`);
      const res = await fetch(
        `https://${zdSettings.zdSubdomain}.zendesk.com/api/v2/tickets/${ticketNum}.json`,
        { headers: { 'Authorization': `Basic ${auth}`, 'Content-Type': 'application/json' } }
      );
      if (res.status === 401) throw new Error('Authentication failed. Check your Zendesk email and API token in Settings.');
      if (res.status === 404) throw new Error(`Ticket #${ticketNum} not found in your Zendesk.`);
      if (!res.ok) throw new Error(`Zendesk error: ${res.status}`);
      const d = await res.json();
      const t = d.ticket;
      setZdResult({
        id: t.id,
        subject: t.subject,
        status: t.status,
        priority: t.priority || 'normal',
        created: t.created_at,
        updated: t.updated_at,
        requester_id: t.requester_id,
        description: t.description?.slice(0, 200),
      });
    } catch (e) {
      setZdError(e.message);
    }
    setZdLoading(false);
  };

  const ZD_STATUS_COLORS = { open: C.red, pending: C.warn, solved: C.accent, closed: C.t3, new: C.blue };

  return (
    <ScrollView style={{ flex: 1 }} contentContainerStyle={ss.toolsContent} keyboardShouldPersistTaps="handled">
      {/* Tool tabs */}
      <View style={ss.toolTabRow}>
        <TouchableOpacity style={[ss.toolTab, toolTab === 'tx' && ss.toolTabOn]} onPress={() => setToolTab('tx')}>
          <Text style={[ss.toolTabTxt, toolTab === 'tx' && ss.toolTabTxtOn]}>⛓️ TX Checker</Text>
        </TouchableOpacity>
        <TouchableOpacity style={[ss.toolTab, toolTab === 'zd' && ss.toolTabOn]} onPress={() => setToolTab('zd')}>
          <Text style={[ss.toolTabTxt, toolTab === 'zd' && ss.toolTabTxtOn]}>🔍 Zendesk</Text>
        </TouchableOpacity>
      </View>

      {/* ── BLOCKCHAIN TX CHECKER ── */}
      {toolTab === 'tx' && (
        <View>
          <Text style={ss.toolTitle}>⛓️ Blockchain TX Checker</Text>
          <Text style={ss.toolDesc}>Paste a customer's TxID to instantly check its confirmation status on the blockchain.</Text>

          <Text style={ss.inputLbl}>SELECT NETWORK</Text>
          <View style={ss.networkRow}>
            {TX_NETWORKS.map(n => (
              <TouchableOpacity key={n} style={[ss.networkBtn, txNetwork === n && ss.networkBtnOn]} onPress={() => setTxNetwork(n)}>
                <Text style={[ss.networkBtnTxt, txNetwork === n && ss.networkBtnTxtOn]}>{n}</Text>
              </TouchableOpacity>
            ))}
          </View>

          <Text style={ss.inputLbl}>TRANSACTION ID (TxID / Hash)</Text>
          <TextInput
            style={ss.toolInput}
            placeholder="Paste TxID here..."
            placeholderTextColor={C.t3}
            value={txid}
            onChangeText={setTxid}
            autoCapitalize="none"
            autoCorrect={false}
          />

          <TouchableOpacity
            style={[ss.toolSubmit, (!txid.trim() || txLoading) && { opacity: 0.5 }]}
            onPress={checkTx} disabled={!txid.trim() || txLoading}>
            {txLoading
              ? <ActivityIndicator color={C.white} />
              : <Text style={ss.toolSubmitTxt}>🔍 Check Transaction</Text>}
          </TouchableOpacity>

          {txError ? (
            <View style={ss.errorBox}>
              <Text style={ss.errorTxt}>⚠️ {txError}</Text>
            </View>
          ) : null}

          {txResult && (
            <View style={ss.resultCard}>
              <View style={ss.resultCardHead}>
                <Text style={ss.resultCardTitle}>Transaction Result</Text>
                <TouchableOpacity style={ss.copyBtn} onPress={copyTxReply}>
                  <Text style={ss.copyBtnTxt}>Copy Reply</Text>
                </TouchableOpacity>
              </View>

              <View style={ss.txRow}>
                <Text style={ss.txLbl}>Network</Text>
                <Text style={ss.txVal}>{txResult.network}</Text>
              </View>
              <View style={ss.txRow}>
                <Text style={ss.txLbl}>Status</Text>
                <Text style={[ss.txVal, { color: txResult.status.includes('✅') ? C.accent : C.warn }]}>{txResult.status}</Text>
              </View>
              <View style={ss.txRow}>
                <Text style={ss.txLbl}>Confirmations</Text>
                <Text style={[ss.txVal, {
                  color: Number(txResult.confirmations) >= txResult.required ? C.accent : C.warn
                }]}>
                  {txResult.confirmations} / {txResult.required} required
                </Text>
              </View>
              <View style={ss.txRow}>
                <Text style={ss.txLbl}>Amount</Text>
                <Text style={ss.txVal}>{txResult.amount}</Text>
              </View>
              <View style={ss.txRow}>
                <Text style={ss.txLbl}>Network Fee</Text>
                <Text style={ss.txVal}>{txResult.fee}</Text>
              </View>
              <View style={[ss.txRow, { borderBottomWidth: 0 }]}>
                <Text style={ss.txLbl}>TxID</Text>
                <Text style={[ss.txVal, { fontSize: 10, flex: 1 }]} numberOfLines={1}>{txResult.txid}</Text>
              </View>

              {/* Confirmation Progress Bar */}
              <View style={ss.confBarWrap}>
                <View style={ss.confBarTrack}>
                  <View style={[ss.confBarFill, {
                    width: `${Math.min(100, (Number(txResult.confirmations) / txResult.required) * 100)}%`,
                    backgroundColor: Number(txResult.confirmations) >= txResult.required ? C.accent : C.warn,
                  }]} />
                </View>
                <Text style={ss.confBarLbl}>

// ─── STATS SCREEN ─────────────────────────────────────────
function StatsScreen({ log }) {
  const stats = {
    total: log.length, used: log.filter(e => e.status === 'Used').length,
    needsMod: log.filter(e => e.status === 'Needs Modification').length,
    ticketed: log.filter(e => e.ticketId?.trim()).length,
  };
  const maxCat = Math.max(1, ...Object.keys(BUILT_IN_TEMPLATES).map(c => log.filter(e => e.category === c).length));

  return (
    <ScrollView style={{ flex: 1 }} contentContainerStyle={ss.statsContent}>
      <View style={ss.statGrid}>
        {[
          { icon: '📨', label: 'Total Used', val: stats.total, color: C.accent },
          { icon: '✅', label: 'Successful', val: stats.used, color: C.accent },
          { icon: '⚠️', label: 'Need Fix', val: stats.needsMod, color: C.warn },
          { icon: '🎫', label: 'Ticketed', val: stats.ticketed, color: C.blue },
        ].map(s => (
          <View key={s.label} style={ss.statCard}>
            <Text style={ss.statIcon}>{s.icon}</Text>
            <Text style={[ss.statVal, { color: s.color }]}>{s.val}</Text>
            <Text style={ss.statLbl}>{s.label}</Text>
          </View>
        ))}
      </View>
      <Text style={ss.secLabel}>CATEGORY USAGE</Text>
      {Object.keys(BUILT_IN_TEMPLATES).map(cat => {
        const count = log.filter(e => e.category === cat).length;
        const pct = (count / maxCat) * 100;
        return (
          <View key={cat} style={ss.barRow}>
            <Text style={ss.barLbl}>{CAT_ICONS[cat]} {cat.split(' ')[0]}</Text>
            <View style={ss.barTrack}><View style={[ss.barFill, { width: `${pct}%` }]} /></View>
            <Text style={ss.barCount}>{count}</Text>
          </View>
        );
      })}
      <Text style={[ss.secLabel, { marginTop: 16 }]}>STATUS BREAKDOWN</Text>
      {STATUS_OPTIONS.map(s => {
        const count = log.filter(e => e.status === s).length;
        const pct = stats.total ? (count / stats.total) * 100 : 0;
        return (
          <View key={s} style={ss.stsRow}>
            <View style={[ss.stsDot, { backgroundColor: STATUS_COLORS[s] }]} />
            <Text style={ss.stsLbl}>{s}</Text>
            <View style={ss.stsTrack}><View style={[ss.stsFill, { width: `${pct}%`, backgroundColor: STATUS_COLORS[s] }]} /></View>
            <Text style={ss.stsCount}>{count}</Text>
          </View>
        );
      })}
      {log.length === 0 && (
        <View style={ss.empty}><Text style={ss.emptyEmoji}>📊</Text>
          <Text style={ss.emptyTitle}>No data yet</Text>
          <Text style={ss.emptyTxt}>Start using templates to see analytics here</Text></View>
      )}
    </ScrollView>
  );
}
